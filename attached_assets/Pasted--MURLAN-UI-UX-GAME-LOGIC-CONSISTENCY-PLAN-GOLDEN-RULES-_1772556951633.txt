// components/MenuLayout.tsx

// Provides: safe-area-aware full screen, scroll container, consistent
// padding, orientation-responsive layout, background color.
// Use this as the outermost wrapper on EVERY menu/lobby screen.

import React from 'react';
import {
  View, ScrollView, StyleSheet, Platform,
  useWindowDimensions, ViewStyle,
} from 'react-native';
import { useSafeAreaInsets } from 'react-native-safe-area-context';
import { Colors, Spacing } from '@/lib/theme';

interface MenuLayoutProps {
  children: React.ReactNode;
  scrollable?: boolean;       // default true
  centered?: boolean;         // default true — centers content vertically
  style?: ViewStyle;
}

export function MenuLayout({ children, scrollable = true, centered = true, style }: MenuLayoutProps) {
  const insets = useSafeAreaInsets();
  const { width, height } = useWindowDimensions();
  const isLandscape = width > height;

  const paddingTop    = Math.max(insets.top    + Spacing.md, Spacing.lg);
  const paddingBottom = Math.max(insets.bottom + Spacing.md, Spacing.lg);
  const paddingLeft   = Math.max(insets.left   + Spacing.md, isLandscape ? Spacing.xl : Spacing.md);
  const paddingRight  = Math.max(insets.right  + Spacing.md, isLandscape ? Spacing.xl : Spacing.md);

  const containerStyle = [
    styles.root,
    { paddingTop, paddingBottom, paddingLeft, paddingRight },
    style,
  ];

  if (!scrollable) {
    return (
      <View style={[styles.root, containerStyle, centered && styles.centered]}>
        {children}
      </View>
    );
  }

  return (
    <ScrollView
      style={styles.root}
      contentContainerStyle={[styles.scroll, containerStyle, centered && styles.centered]}
      keyboardShouldPersistTaps="handled"
      showsVerticalScrollIndicator={false}
    >
      {children}
    </ScrollView>
  );
}

const styles = StyleSheet.create({
  root:     { flex: 1, backgroundColor: Colors.bg },
  scroll:   { flexGrow: 1 },
  centered: { justifyContent: 'center', alignItems: 'center' },
});

import React from 'react';
import {
  TouchableOpacity, Text, StyleSheet,
  ViewStyle, ActivityIndicator, Platform,
} from 'react-native';
import { Colors, Spacing, Radius, FontSize, Shadow } from '@/lib/theme';

type Variant = 'primary' | 'secondary' | 'danger' | 'ghost';

interface MenuButtonProps {
  label: string;
  onPress: () => void;
  variant?: Variant;
  disabled?: boolean;
  loading?: boolean;
  icon?: React.ReactNode;   // optional icon left of label
  style?: ViewStyle;
  fullWidth?: boolean;      // default true
}

export function MenuButton({
  label, onPress, variant = 'primary',
  disabled, loading, icon, style, fullWidth = true,
}: MenuButtonProps) {
  const isDisabled = disabled || loading;

  return (
    <TouchableOpacity
      style={[
        styles.base,
        styles[variant],
        fullWidth && styles.fullWidth,
        isDisabled && styles.disabled,
        style,
      ]}
      onPress={onPress}
      disabled={isDisabled}
      activeOpacity={0.75}
      accessibilityRole="button"
      accessibilityLabel={label}
      accessibilityState={{ disabled: isDisabled }}
    >
      {loading
        ? <ActivityIndicator color={variant === 'primary' ? Colors.bg : Colors.gold} />
        : <>
            {icon && <>{icon}</>}
            <Text style={[styles.label, styles[`${variant}Label` as any]]}>{label}</Text>
          </>
      }
    </TouchableOpacity>
  );
}

// ─── Exchange Phase ─────────────────────────────────────────────────────────────

export interface ExchangeResult {
  winnerId: string | number;
  winnerName: string;
  loserId: string | number;
  loserName: string;
  cardGivenByLoser: Card;    // best card — taken automatically
  cardGivenByWinner: Card;   // winner chose this (3–10)
  bothJokersException: boolean;
}

// Given the final ranking of a completed game, compute who exchanges with whom.
// Returns null if only 2 players with no exchange needed (edge case check).
// MUST match the exchange rules in replit.md exactly:
//   Loser gives best card to winner. Winner gives back a 3–10 card.
//   Both jokers exception: no exchange, winner starts.
export function computeExchangePairs(
  rankedPlayers: Player[]  // index 0 = winner, last index = loser
): Array<{ winner: Player; loser: Player }> {
  if (rankedPlayers.length < 2) return [];
  // 2 players: 1 pair. 4 players: winner↔loser, 2nd↔3rd
  const pairs: Array<{ winner: Player; loser: Player }> = [];
  const half = Math.floor(rankedPlayers.length / 2);
  for (let i = 0; i < half; i++) {
    pairs.push({
      winner: rankedPlayers[i],
      loser: rankedPlayers[rankedPlayers.length - 1 - i],
    });
  }
  return pairs;
}

// Add to components/GameShared.tsx (or extract to lib/useTurnPulse.ts)

import {
  useSharedValue, useAnimatedStyle, withRepeat,
  withSequence, withTiming, interpolate,
} from 'react-native-reanimated';

export function useTurnPulse(isMyTurn: boolean) {
  const pulse = useSharedValue(0);

  React.useEffect(() => {
    if (isMyTurn) {
      pulse.value = withRepeat(
        withSequence(
          withTiming(1, { duration: ANIMATION.TURN_PULSE_PERIOD_MS / 2 }),
          withTiming(0, { duration: ANIMATION.TURN_PULSE_PERIOD_MS / 2 }),
        ),
        -1,   // repeat forever
        false // no reverse (handled by sequence)
      );
    } else {
      pulse.value = withTiming(0, { duration: 300 });
    }
  }, [isMyTurn]);

  // Animated style for the HAND CONTAINER (glow effect)
  const glowStyle = useAnimatedStyle(() => {
    const opacity = interpolate(pulse.value, [0, 1], [0.45, 1.0]);
    const radius  = interpolate(pulse.value, [0, 1], [12, 28]);
    if (Platform.OS === 'web') {
      return {
        // Web: boxShadow with animated values
        boxShadow: `0 0 ${radius}px rgba(201,168,76,${opacity})`,
      } as any;
    }
    return {
      shadowColor: Colors.gold,
      shadowOffset: { width: 0, height: 0 },
      shadowOpacity: opacity,
      shadowRadius: radius,
      elevation: interpolate(pulse.value, [0, 1], [8, 20]),
    };
  });

  // Animated style for the CARD ROW (subtle scale zoom)
  const scaleStyle = useAnimatedStyle(() => {
    const scale = interpolate(pulse.value, [0, 1], [1.0, 1.035]);
    return { transform: [{ scale }] };
  });

  return { glowStyle, scaleStyle };
}
